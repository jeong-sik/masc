let home_fixture = ref None
let checks = ref 0
let string = ()
let check _ label expected actual =
  incr checks;
  if not (String.equal expected actual) then
    failwith (Printf.sprintf "%s: expected %S, actual %S" label expected actual)

module Baseline = struct
module Common = struct let masc_dirname = ".masc" end
let raw_value_opt = function
  | "HOME" -> !home_fixture
  | name -> failwith ("unexpected environment fixture lookup: " ^ name)
let trim_opt = function
  | Some raw ->
      let trimmed = String.trim raw in
      if trimmed = "" then None else Some trimmed
  | None -> None

let strip_path_trailing_slashes value =
  let trimmed = String.trim value in
  let rec loop current =
    let len = String.length current in
    if len > 1 && current.[len - 1] = '/' then
      loop (String.sub current 0 (len - 1))
    else
      current
  in
  if trimmed = "" then "" else loop trimmed

let expand_home_prefix value =
  if String.length value >= 2 && value.[0] = '~' && value.[1] = '/' then
    match raw_value_opt "HOME" |> trim_opt with
    | Some home -> Filename.concat home (String.sub value 2 (String.length value - 2))
    | None -> value
  else
    value

let normalize_path_lexically value =
  let value = expand_home_prefix (String.trim value) in
  if value = "" then ""
  else
    let absolute = value.[0] = '/' in
    let parts = String.split_on_char '/' value in
    let rec fold acc = function
      | [] -> acc
      | "" :: rest | "." :: rest -> fold acc rest
      | ".." :: rest -> (
          match acc, absolute with
          | _ :: acc_tail, _ -> fold acc_tail rest
          | [], true -> fold [] rest
          | [], false -> fold [ ".." ] rest)
      | part :: rest -> fold (part :: acc) rest
    in
    let normalized_parts = List.rev (fold [] parts) in
    match absolute, normalized_parts with
    | true, [] -> "/"
    | true, parts -> "/" ^ String.concat "/" parts
    | false, [] -> "."
    | false, parts -> String.concat "/" parts

let normalize_masc_base_path_input path =
  let normalized = path |> normalize_path_lexically |> strip_path_trailing_slashes in
  if normalized = "" then ""
  else if String.equal (Filename.basename normalized) Common.masc_dirname then
    match Filename.dirname normalized with
    | "" -> "."
    | parent -> parent
  else
    normalized


end

module Candidate = struct
module Common = struct let masc_dirname = ".masc" end
let raw_value_opt = function

let trim_opt = function
  | Some raw ->
      let trimmed = String.trim raw in
      if trimmed = "" then None else Some trimmed
  | None -> None

let strip_path_trailing_slashes value =
  let trimmed = String.trim value in
  let rec loop current =
    let len = String.length current in
    if len > 1 && current.[len - 1] = '/' then
      loop (String.sub current 0 (len - 1))
    else
      current
  in
  if trimmed = "" then "" else loop trimmed

let expand_home_prefix value =
  if String.length value >= 2 && value.[0] = '~' && value.[1] = '/' then
    match raw_value_opt "HOME" |> trim_opt with
    | Some home -> Filename.concat home (String.sub value 2 (String.length value - 2))
    | None -> value
  else
    value

type lexical_path_form = Normal_form | Needs_normalization

let rec classify_path_components value length component_start index =
  if index < length && value.[index] <> '/' then
    classify_path_components value length component_start (index + 1)
  else
    let component_length = index - component_start in
    if component_length = 0
       || (component_length = 1 && value.[component_start] = '.')
       || (component_length = 2 && value.[component_start] = '.'
           && value.[component_start + 1] = '.')
    then Needs_normalization
    else if index = length then Normal_form
    else classify_path_components value length (index + 1) (index + 1)

let classify_lexical_path value =
  let length = String.length value in
  match length with
  | 0 -> Needs_normalization
  | 1 when value.[0] = '/' -> Normal_form
  | _ ->
      let first_component = if value.[0] = '/' then 1 else 0 in
      classify_path_components value length first_component first_component

let normalize_path_lexically value =
  let value = expand_home_prefix (String.trim value) in
  if value = "" then ""
  else
    match classify_lexical_path value with
    | Normal_form -> value
    | Needs_normalization ->
        let absolute = value.[0] = '/' in
        let parts = String.split_on_char '/' value in
        let rec fold acc = function
          | [] -> acc
          | "" :: rest | "." :: rest -> fold acc rest
          | ".." :: rest -> (
              match acc, absolute with
              | _ :: acc_tail, _ -> fold acc_tail rest
              | [], true -> fold [] rest
              | [], false -> fold [ ".." ] rest)
          | part :: rest -> fold (part :: acc) rest
        in
        let normalized_parts = List.rev (fold [] parts) in
        match absolute, normalized_parts with
        | true, [] -> "/"
        | true, parts -> "/" ^ String.concat "/" parts
        | false, [] -> "."
        | false, parts -> String.concat "/" parts

let normalize_masc_base_path_input path =
  let normalized = path |> normalize_path_lexically |> strip_path_trailing_slashes in
  if normalized = "" then ""
  else if String.equal (Filename.basename normalized) Common.masc_dirname then
    match Filename.dirname normalized with
    | "" -> "."
    | parent -> parent
  else
    normalized


end

  check string "lexical differential"
    (Baseline.normalize_path_lexically path) (Candidate.normalize_path_lexically path);
  check string "base-input differential"
    (Baseline.normalize_masc_base_path_input path)
    (Candidate.normalize_masc_base_path_input path)

  let rec enumerate remaining prefix =
    compare prefix;

        enumerate (remaining - 1) (prefix ^ String.make 1 byte)) alphabet
