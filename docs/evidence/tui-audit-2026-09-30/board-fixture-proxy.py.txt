import select
import socket
import socketserver
import sys

port = int(sys.argv[1])

class Forward(socketserver.BaseRequestHandler):
    def handle(self):
        with socket.create_connection(('host.docker.internal', port), timeout=10) as upstream:
            peers = [self.request, upstream]
            while True:
                ready, _, _ = select.select(peers, [], [])
                for peer in ready:
                    data = peer.recv(65536)
                    if not data:
                        return
                    (upstream if peer is self.request else self.request).sendall(data)

class Server(socketserver.ThreadingTCPServer):
    allow_reuse_address = True
    daemon_threads = True

with Server(('127.0.0.1', port), Forward) as server:
    server.serve_forever()
