let draw ~korean ~inner_width =
  let headers =
    if korean then
      { Run_table.lhead_started = "\xec\x8b\x9c\xec\x9e\x91"
      ; lhead_subject = "\xec\x9a\x94\xec\xb2\xad\xec\x9e\x90"
      ; lhead_status = "\xec\x83\x81\xed\x83\x9c"
      ; lhead_elapsed = "\xea\xb1\xb8\xeb\xa6\xb0 \xec\x8b\x9c\xea\xb0\x84"
      ; lhead_slot = "\xec\x8a\xac\xeb\xa1\xaf"
      }
    else
      { Run_table.lhead_started = "Started"
      ; lhead_subject = "Actor"
      ; lhead_status = "Status"
      ; lhead_elapsed = "Elapsed"
      ; lhead_slot = "Slot"
      }
  in
  let table =
    Run_table.lane_run_table ~headers ~subject:Run_table.Run_actor_name
      ~subjects:(List.map (fun (_, subject, _, _, _, _) -> subject) rows)
      ~inner_width
  in
  Printf.printf "-- inner width %d, %s\n" inner_width (if korean then "ko" else "en");
  print_endline ("  " ^ Run_table.lane_run_header_row table);
  List.iter
    (fun (started, subject, ko, en, elapsed, slot) ->
      print_endline
        ("  "
        ^ Run_table.lane_run_row table ~now ~status_style:""
            { Run_table.lrow_started = started
            ; lrow_subject = subject
            ; lrow_status = (if korean then ko else en)
            ; lrow_elapsed = elapsed
            ; lrow_slot = slot
            }))
    rows;
  print_newline ()

let () =
  draw ~korean:true ~inner_width:196;
  draw ~korean:false ~inner_width:116;
  draw ~korean:true ~inner_width:76;
  draw ~korean:true ~inner_width:56
