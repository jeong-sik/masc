def canonical_queue_source_bytes(raw, expected):
    """Select pending[0].source lexical bytes; never reserialize float values.

    The frozen codec uses the same stimulus serializer for the snapshot and
    source ref. Restrict this proof to its controlled ASCII, finite source.
    """
    def unique_object(pairs):
        result = {}
        for key, value in pairs:
            if key in result:
                raise Incomplete('duplicate JSON member in source-address evidence')
            result[key] = value
        return result
    def reject_constant(value):
        raise Incomplete('nonfinite JSON token in source-address evidence')
    decoder = json.JSONDecoder(object_pairs_hook=unique_object, parse_constant=reject_constant)
    text = raw.decode('utf-8')
    def skip(index):
        while index < len(text) and text[index] in ' \t\r\n':
            index += 1
        return index
    decoded, end = decoder.raw_decode(text, skip(0))
    if skip(end) != len(text) or decoded != expected or len(decoded['pending']) != 1:
        raise Incomplete('source-address snapshot differs from final single-source observation')
    def member(start, key):
        index = skip(start)
        if text[index] != '{':
            raise Incomplete('source-address JSON path is not an object')
        index = skip(index + 1)
        found = []
        while text[index] != '}':
            name, next_index = decoder.raw_decode(text, index)
            index = skip(next_index)
            if not isinstance(name, str) or text[index] != ':':
                raise Incomplete('invalid source-address JSON member')
            start_value = skip(index + 1)
            _, end_value = decoder.raw_decode(text, start_value)
            if name == key:
                found.append((start_value, end_value))
            index = skip(end_value)
            if text[index] == '}':
                break
            if text[index] != ',':
                raise Incomplete('invalid source-address JSON separator')
            index = skip(index + 1)
        if len(found) != 1:
            raise Incomplete('ambiguous or absent source-address JSON path')
        return found[0]
    pending, _ = member(skip(0), 'pending')
    if text[pending] != '[':
        raise Incomplete('source-address pending is not an array')
    entry = skip(pending + 1)
    _, entry_end = decoder.raw_decode(text, entry)
    if text[skip(entry_end)] != ']':
        raise Incomplete('source-address evidence has multiple pending entries')
    start, end = member(entry, 'source')
    source = decoder.raw_decode(text, start)[0]
    def controlled(value):
        if isinstance(value, dict):
            return all(key.isascii() and controlled(item) for key, item in value.items())
        if isinstance(value, list):
            return all(controlled(item) for item in value)
        if isinstance(value, str):
            return value.isascii()
        if isinstance(value, float):
            return math.isfinite(value)
        return value is None or type(value) in (int, bool)
    if source != expected['pending'][0]['source'] or not controlled(source):
        raise Incomplete('source-address bytes are outside controlled finite ASCII scope')
    lexical = text[start:end].encode('utf-8')
    compact, quoted, escaped = bytearray(), False, False
    for byte in lexical:
        if quoted:
            if escaped:
                escaped = False
            elif byte == 92:
                escaped = True
            elif byte == 34:
                quoted = False
        else:
            if byte in (32, 9, 13, 10):
                continue
            if byte == 34:
                quoted = True
        compact.append(byte)
    if quoted or escaped:
        raise Incomplete('unterminated source-address JSON string')
    return bytes(compact)

            setup_raw, setup_state = queue_snapshot()
            witnesses = [entry['witness'] for entry in setup_state['projected_dispositions'] if entry['storage'] == 'compact_witness'
                         and entry['witness']['post_id'] == setup_post]
            check('exact setup cancellation terminal and authoritative queue empty', empty(setup_state) and len(witnesses) == 1
                  and witnesses[0]['disposition']['kind'] == 'cancel'
                  and witnesses[0]['source_ref'] == exact[0]['detail']['source_ref']
                  and str(witnesses[0]['source_incarnation']) == str(exact[0]['detail']['source_incarnation']))
            save_raw('setup-cancelled-queue.json', setup_raw)
            check('canonical setup produced an empty regular WAL', stat.S_ISREG(wal.lstat().st_mode) and wal.read_bytes() == b'')
            save_raw('setup-cancelled-wal.jsonl', b'')
            ev.json('setup-cancelled-inventory.json', inventory('setup-cancelled-inventory'))
            original = create('audit-settle-original', 15)
            before_health = health('before-original-lock')
            base_counts = before_health['schedule_runner']['totals']
            # No open/close/read_bytes/copytree of this WAL while the parent record lock is held.
            wal_fd = os.open(wal, os.O_RDWR | os.O_CLOEXEC | os.O_NOFOLLOW)
            fcntl.lockf(wal_fd, fcntl.LOCK_EX | fcntl.LOCK_NB, 0, 0, os.SEEK_SET)
            check('canonical WAL lock acquired before original due', time.time() < original['due_at'])
            held_stat = os.fstat(wal_fd)
            check('locked canonical WAL still empty', held_stat.st_size == 0)
            deadline = time.monotonic() + a.wait_seconds
            with contextlib.closing(select.kqueue()) as queue:
                dfd, ffd, identity = os.open(primary.parent, os.O_EVTONLY), None, None
                try:
                    notes = select.KQ_NOTE_WRITE | select.KQ_NOTE_RENAME | select.KQ_NOTE_DELETE | select.KQ_NOTE_ATTRIB
                    queue.control([select.kevent(dfd, filter=select.KQ_FILTER_VNODE, flags=select.KQ_EV_ADD | select.KQ_EV_CLEAR, fflags=notes)], 0, 0)
                    while True:
                        if first.poll() is not None or time.monotonic() >= deadline:
                            raise Incomplete('single Running/consumer barrier trial missed; no automatic retrial')
                        current_stat = primary.stat()
                        if identity != (current_stat.st_dev, current_stat.st_ino):
                            if ffd is not None:
                                os.close(ffd)
                            ffd = os.open(primary, os.O_EVTONLY)
                            identity = (current_stat.st_dev, current_stat.st_ino)
                            queue.control([select.kevent(ffd, filter=select.KQ_FILTER_VNODE, flags=select.KQ_EV_ADD | select.KQ_EV_CLEAR, fflags=notes)], 0, 0)
                        before_raw, before = read_ledger()
                        candidate = running(before)
                        if candidate is not None and not lease.exists():
                            os.kill(first.pid, signal.SIGSTOP)
                            stopped.add(first.pid)
                            break
                        events = queue.control(None, 8, min(1, max(0, deadline - time.monotonic())))
                        ev.record({'kind': 'kqueue', 'observed_at': time.time(), 'events': [{'ident': e.ident, 'filter': e.filter, 'fflags': e.fflags} for e in events]})
                    while True:
                        pid, status = os.waitpid(first.pid, os.WUNTRACED | os.WNOHANG)
                        if pid:
                            if not os.WIFSTOPPED(status):
                                first.returncode = os.waitstatus_to_exitcode(status)
                                raise Incomplete('owned process exited instead of stopping')
                            check('kernel confirms genuine SIGSTOP', os.WSTOPSIG(status) == signal.SIGSTOP)
                            ev.json('stopped-process.json', {'pid': pid, 'waitpid_status': status, 'at': time.time()})
                            break
                        if time.monotonic() >= deadline:
                            raise Incomplete('SIGSTOP receipt observation incomplete')
                        time.sleep(.01)
                    frozen_raw, frozen = read_ledger()
                    frozen_wake = running(frozen)
                    save_raw('before-stop-ledger.json', before_raw)
                    save_raw('frozen-ledger.json', frozen_raw)
                    check('frozen exact genuine Running attempt', frozen_wake is not None and frozen_wake['attempt_id'] == candidate['attempt_id'])
                    qraw, qs = queue_snapshot()
                    current_wal = wal.lstat()
                    check('publication has not occurred and barrier inode unchanged', qraw == setup_raw and empty(qs)
                          and (current_wal.st_dev, current_wal.st_ino) == (held_stat.st_dev, held_stat.st_ino)
                          and os.fstat(wal_fd).st_size == 0 and os.pread(wal_fd, 1, 0) == b'' and not lease.exists())
                    save_raw('frozen-queue.json', qraw)
                    save_raw('frozen-wal.jsonl', os.pread(wal_fd, os.fstat(wal_fd).st_size, 0))
                    ev.json('wal-barrier.json', {'device': held_stat.st_dev, 'inode': held_stat.st_ino,
                            'held_until_after_faults': True, 'opened_wal_once': True, 'at': time.time()})
                    check('owned queue directory is real', queue_dir.is_dir() and not queue_dir.is_symlink())
                    original_mode = stat.S_IMODE(queue_dir.stat().st_mode)
                    check('original owner directory permits read/search/write', original_mode & 0o700 == 0o700)
                    os.chmod(queue_dir, original_mode & ~0o222)
                    scratch = queue_dir / 'controller-publication-refusal-scratch'
                    try:
                        probe_fd = os.open(scratch, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
                    except PermissionError as error:
                        ev.json('queue-effective-write-refusal.json', {'errno': error.errno, 'original_mode': oct(original_mode),
                                'fault_mode': oct(stat.S_IMODE(queue_dir.stat().st_mode)), 'at': time.time()})
                    else:
                        os.close(probe_fd)
                        scratch.unlink()
                        raise Incomplete('effective write was not refused; publication fault premise absent')
                    check('queue remains readable under publication fault', queue_snapshot()[0] == setup_raw)
                    check('primary staging path absent after Running primary commit', not staging.exists())
                    staging.mkdir(mode=0o700)
                    ev.json('faults-installed.json', {'pid': first.pid, 'at': time.time(), 'queue_mode': oct(stat.S_IMODE(queue_dir.stat().st_mode)),
                            'ledger_staging_directory': str(staging), 'no_domain_bytes_authored': True})
                    release_wal()
                    resumed_at = time.time()
                    os.kill(first.pid, signal.SIGCONT)
                    stopped.discard(first.pid)
                    ev.json('resumed-process.json', {'pid': first.pid, 'at': resumed_at})
                finally:
                    if ffd is not None:
                        os.close(ffd)
                    os.close(dfd)
            def first_failure(value):
                h, ledger, qs = value
                status = h['schedule_runner']
                counts = status.get('last_counts') or {}
                return (completed(status) and status['pending_settlement_count'] == 1 and status['status'] == 'degraded'
                    and counts.get('dispatch_failed') == 1 and counts.get('dispatch_start_rejected') == 0
                    and status['totals']['dispatch_failed'] == base_counts['dispatch_failed'] + 1
                    and status['totals']['dispatch_start_rejected'] == base_counts['dispatch_start_rejected']
                    and bool(status['settlement_errors']) and running(ledger) == frozen_wake and empty(qs))
            failed = wait(lambda: observed('first-failure'), first_failure, process=first)
            failure_status = failed[0]['schedule_runner']
            ev.json('first-failure-health.json', failed[0])
            log_at_failure = (output / 'server-A.log').read_bytes()
            save_raw('server-at-first-failure.log', log_at_failure)
            check('new failed dispatch retained as pending; not rejected start', first_failure(failed))
            queue_restored_at = time.time()
            os.chmod(queue_dir, original_mode)
            check('queue-only repair preserves real ledger fault', staging.is_dir() and empty(queue_snapshot()[1]))
            ev.json('queue-repaired.json', {'at': queue_restored_at, 'restored_mode': oct(original_mode), 'pid': first.pid})
            def replay_still_pending(value):
                h, ledger, qs = value
                status = h['schedule_runner']
                return (completed(status) and status['tick_count'] > failure_status['tick_count']
                    and status['last_tick_started_at'] > queue_restored_at and status['status'] == 'degraded'
                    and status['pending_settlement_count'] == 1 and bool(status['settlement_errors'])
                    and status['last_counts']['dispatch_failed'] == 0 and status['last_counts']['dispatch_succeeded'] == 0
                    and status['totals'] == failure_status['totals'] and running(ledger) == frozen_wake and empty(qs))
            pending = wait(lambda: observed('queue-repaired-ledger-fault'), replay_still_pending, process=first)
            ev.json('queue-repaired-completed-cadence-health.json', pending[0])
            check('completed later cadence retries only original settlement with repaired empty queue', replay_still_pending(pending))
            staging.rmdir()
            ledger_restored_at = time.time()
            ev.json('ledger-repaired.json', {'at': ledger_restored_at, 'pid': first.pid})
            def retry_settled(value):
                h, ledger, qs = value
                status = h['schedule_runner']

                return (completed(status) and status['last_tick_started_at'] > ledger_restored_at
                    and status['pending_settlement_count'] == 0 and status['settlement_errors'] == []
                    and status['totals'] == failure_status['totals'] and status['last_counts']['dispatch_failed'] == 0
                    and status['last_counts']['dispatch_succeeded'] == 0 and row(ledger, original['schedule_id'])['status'] == 'due'
                    and len(wakes(ledger, original)) == 1 and len(old) == 1 and old[0]['status'] == 'failed'

                    and failure_status['last_tick_started_at'] <= old[0]['finished_at'] <= failure_status['last_tick_finished_at']
                    and old[0]['finished_at'] < queue_restored_at < ledger_restored_at and empty(qs))
            settled = wait(lambda: observed('retry-settled-only'), retry_settled, process=first)
            ev.json('retry-settled-only-health.json', settled[0])

            old_wake = wakes(settled[1], original)[0]

                    and status['pending_settlement_count'] == 0 and row(ledger, original['schedule_id'])['status'] == 'succeeded'
                    and len(attempt_wakes) == 2 and len(new) == 1 and new[0]['settlement_kind'] == 'accepted'
                    and new[0]['status'] == 'succeeded' and [w for w in attempt_wakes if w['attempt_id'] == frozen_wake['attempt_id']] == [old_wake]

            cleanup_errors = []
            def cleanup(label, action):
                try:
                    action()
                except Exception as error:
                    cleanup_errors.append({'step': label, 'kind': type(error).__name__, 'message': str(error)})
            if original_mode is not None:

            for label, process in reversed(list(processes.items())):

                    if process.pid in stopped:
                        cleanup('continue-own-stopped-process', lambda: os.kill(process.pid, signal.SIGCONT))
                        stopped.discard(process.pid)
                    cleanup('terminate-own-process', process.terminate)
                    try:
                        process.wait(timeout=a.wait_seconds)
                    except subprocess.TimeoutExpired:
                        process.kill()
                        process.wait()
                        final_forced_kills.append({'label': label, 'pid': process.pid})
            if primary is not None and primary.exists():
                save_raw('after-owned-shutdown-ledger.json', primary.read_bytes())
            ev.json('summary.json', {'outcome': outcome, 'checks': checks,
                'cleanup_errors': cleanup_errors, 'final_cleanup_forced_kills': final_forced_kills, 'process_exit_codes': {k: v.returncode for k, v in processes.items()},
                'observer_deadline_means': 'incomplete observation, not producer terminal or product timing failure',
                'restart_count': 0, 'not_proven': provenance['not_proven']})
            ev.json('provenance.json', provenance)
    return 0 if outcome == 'bounded_checks_passed_requires_two_fault_cause_review' and not final_forced_kills and not cleanup_errors and all(p.returncode == 0 for p in processes.values()) else 1
