def sanitize(value):
    # Never retain control or authentication tokens from live observation APIs.
    if isinstance(value, dict):
        return {k: sanitize(v) for k, v in value.items() if k not in
                {'chat_control_token', 'interrupt_token', 'authorization', 'token',
                 'access_token', 'refresh_token', 'bearer_token', 'mcp_session_id'}}
    if isinstance(value, list):
        return [sanitize(v) for v in value]
    if isinstance(value, str):
        # MCP may wrap the same structured response in a JSON text block.
        try:
            decoded = json.loads(value)
        except (json.JSONDecodeError, ValueError):
            return value
        if isinstance(decoded, (dict, list)):
            return json.dumps(sanitize(decoded), ensure_ascii=False)
    return value

def request(path, token=None, body=None):
    headers = {'Content-Type': 'application/json'}
    if token:
        headers['Authorization'] = 'Bearer ' + token
    req = urllib.request.Request(url + path, headers=headers,
          data=None if body is None else json.dumps(body).encode())
    try:
        with urllib.request.urlopen(req, timeout=20) as response:
            status, data = response.status, json.load(response)
    except urllib.error.HTTPError as error:
        status = error.code
        raw = error.read().decode()
        try:
            data = json.loads(raw)
        except json.JSONDecodeError:
            data = {'body': raw}
    records.append({'stage': stage, 'path': path, 'body': body, 'status': status, 'response': sanitize(data)})
    save()
    return status, data

def mcp_roundtrip(token, payload, session=None):
    headers = {'Content-Type': 'application/json', 'Accept': 'application/json, text/event-stream',
               'Authorization': 'Bearer ' + token, 'Mcp-Protocol-Version': '2025-11-25'}
    if session:
        headers['Mcp-Session-Id'] = session
    req = urllib.request.Request(url + '/mcp', headers=headers, data=json.dumps(payload).encode())
    try:
        response = urllib.request.urlopen(req, timeout=30)
    except urllib.error.HTTPError as error:
        response = error
    with response:
        raw = response.read().decode()
        try:
            data = json.loads(raw)
        except json.JSONDecodeError:
            events = [json.loads(line[5:].strip()) for line in raw.splitlines() if line.startswith('data:')]
            data = next((event for event in events if event.get('id') == payload.get('id')), {'raw': raw})
        records.append({'stage': stage, 'path': '/mcp', 'body': payload, 'status': response.status, 'response': sanitize(data)})
        save()
        return response.status, data, response.headers.get('Mcp-Session-Id')


def tool(name, body, token=None):
    global request_id
    selected = admin_token if token is None else token
    if selected not in sessions:
        request_id += 1
        status, result, session = mcp_roundtrip(selected, {'jsonrpc': '2.0', 'id': request_id,
            'method': 'initialize', 'params': {'protocolVersion': '2025-11-25',
            'capabilities': {}, 'clientInfo': {'name': 'schedule-api-audit', 'version': '1'}}})
        if status != 200 or not session or 'error' in result:
            raise RuntimeError('MCP initialize failed: ' + json.dumps(result))
        sessions[selected] = session
    request_id += 1
    status, result, _ = mcp_roundtrip(selected, {'jsonrpc': '2.0', 'id': request_id,
        'method': 'tools/call', 'params': {'name': name, 'arguments': body}}, sessions[selected])
    if 'error' in result:
        data = {'isError': True, 'data': result['error']}
    else:
        reply = result.get('result', {})
        payload = reply.get('structuredContent')
        if payload is None:
            blocks = reply.get('content', [])
            text = next((block['text'] for block in blocks if block.get('type') == 'text'), '')
            try:
                payload = json.loads(text)
            except json.JSONDecodeError:
                payload = {'text': text}
        data = {'isError': reply.get('isError', False), 'data': payload}
    print(name, status, 'isError=' + str(data['isError']), flush=True)
    return status, data


def unwrap(result):
    if isinstance(result, dict) and isinstance(result.get('data'), dict):
        return result['data']
    if isinstance(result, dict) and isinstance(result.get('message'), str):
        try:
            decoded = json.loads(result['message'])
            return decoded if isinstance(decoded, dict) else result
        except json.JSONDecodeError:
            return result
    return result

def table(name,values):
    rows=['['+name+']']

        if isinstance(value,dict):rows.append(table(name+'.'+key,value))
    return '\n'.join(rows)+'\n'

def read_jsonl_files(root):
    rows, sources = [], []
    for path in sorted(root.rglob('*.jsonl')):
        raw = path.read_bytes()
        # A concurrent final append may not be complete. Do not invent a row.
        if raw and not raw.endswith(b'\n'):
            raise Incomplete('concurrent partial JSONL observation: '+str(path))
        values = [json.loads(line) for line in raw.splitlines() if line.strip()]
        rows.extend(values)
        sources.append({'path': str(path), 'sha256': hashlib.sha256(raw).hexdigest(),
                        'bytes': len(raw), 'rows': len(values)})
    return rows, sources

def decode_tool_output(value):
    if isinstance(value, str):
        try:
            return json.loads(value)
        except json.JSONDecodeError:
            return None
    return value if isinstance(value, dict) else None


def checkpoint_selection(path):
    raw = path.read_bytes(); digest = hashlib.sha256(raw).hexdigest()
    if digest in seen_checkpoints:
        return
    cp = json.loads(raw)
    if not isinstance(cp.get('messages'), list):
        return
    uses, results = {}, {}
    for message in cp['messages']:
        content = message.get('content', [])
        if not isinstance(content, list):
            continue
        for block in content:
            if not isinstance(block, dict):
                continue
            if block.get('type') == 'tool_use' and block.get('name') == 'masc_schedule_note_add':
                call = block.get('input') or {}
                if call.get('schedule_id') in [schedule_id, second_id]:
                    uses[block['id']] = call
            elif block.get('type') == 'tool_result' and block.get('is_error') is False:
                data = decode_tool_output(block.get('content'))
                if isinstance(data, dict) and isinstance(data.get('note'), dict):
                    note = data['note']
                    if note.get('schedule_id') in [schedule_id, second_id]:
                        results[block['tool_use_id']] = note
    pairs = [{'tool_use_id': key, 'input': uses[key], 'note': note}
             for key, note in results.items() if key in uses]
    seen_checkpoints.add(digest)
    checkpoint_observations.append({'path': str(path), 'sha256': digest, 'bytes': len(raw),
        'observed_at_unix': time.time(), 'session_id': cp.get('session_id'),
        'turn_count': cp.get('turn_count'), 'created_at': cp.get('created_at'),
        'selected_note_tool_results': pairs,
        'scope': 'Selected persisted lineage facts only; full messages/reasoning are not copied.'})

def final_proof(a, b, aq, bq):
    state, primary_source = primary()
    an, bn = primary_notes(state, a), primary_notes(state, b)
    notes = [[n for n in ns if n.get('body') == body]
             for ns, body in [(an, marker), (bn, second_marker), (an, final_marker)]]
    check('A1 B A2 each exist exactly once with the actual Keeper author',
          all(len(ns) == 1 and ns[0].get('author_id') == keeper for ns in notes),
          {'notes': notes, 'primary': primary_source})
    rows, files = reaction_read(); aid, bid = aq['post_id'], bq['post_id']
    starts = event_times(rows, aid, 'turn_started')
    checkpoints = event_times(rows, aid, 'turn_finished', 'checkpointed')
    bstart = event_times(rows, bid, 'turn_started')
    back = event_times(rows, bid, 'event_queue_ack')
    aack = event_times(rows, aid, 'event_queue_ack')
    arrival = [r['recorded_at_unix'] for r in rows if r.get('stimulus_id') == bid and r.get('record_kind') == 'stimulus']
    if not (starts and checkpoints and bstart and back and aack and arrival):
        raise Incomplete('effects completed but the required after-effect checkpoint/ACK witnesses are absent; sequential completion is not PASS')
    resumed = [s for s in starts if s > max(back)]
    check('A1 precedes B arrival, actual A checkpoint and B source entry',
          any(st < notes[0][0]['created_at'] <= seen_a1_at < ar < cp < bs
              for st in starts for ar in arrival for cp in checkpoints for bs in bstart))
    check('original A has no premature ACK and is readmitted after B ACK',
          min(aack) > max(back) and bool(resumed))
    retentions = [(q, e) for q in queue_observations for e in q['pending']
                  if e['post_id'] == aid and e['checkpoint_retentions'] > 0
                  and e['payload']['schedule_instance_id'] == a['schedule_instance_id']
                  and any(cp <= q['observed_at_unix'] < min(back) for cp in checkpoints)]
    if not retentions:
        raise Incomplete('actual retained A incarnation was not captured')
    incarnations = {e['admitted_revision'] for _, e in retentions}
