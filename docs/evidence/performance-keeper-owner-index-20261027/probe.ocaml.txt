module Keeper_owner_reducer = struct type projection = int option end
module Keeper_owner = struct
  type t = { state : Keeper_owner_reducer.projection Atomic.t }
  let projection owner = Atomic.get owner.state
end
module Keeper_registry_types = struct
  let canonical_base_path_exn value = value
end
module Set_util = struct
module StringMap = Map.Make (String)
end

module Baseline = struct
type lookup =
  | Owner_absent
  | Owner_projection of Keeper_owner_reducer.projection

let owners : (string * Keeper_owner.t) list Atomic.t = Atomic.make []

let cells_mu = Stdlib.Mutex.create ()

let key ~base_path ~keeper_name =
  Keeper_registry_types.canonical_base_path_exn base_path ^ "\000" ^ keeper_name
;;

let install ~base_path ~keeper_name owner =
  let key = key ~base_path ~keeper_name in
  Stdlib.Mutex.protect cells_mu (fun () ->
    let without_key =
      List.filter (fun (candidate, _) -> not (String.equal candidate key)) (Atomic.get owners)
    in
    Atomic.set owners ((key, owner) :: without_key))
;;

let remove ~base_path ~keeper_name owner =
  let key = key ~base_path ~keeper_name in
  Stdlib.Mutex.protect cells_mu (fun () ->
    Atomic.set
      owners
      (List.filter
         (fun (candidate, current) ->
            not (String.equal candidate key && current == owner))
         (Atomic.get owners)))
;;

let lookup ~base_path ~keeper_name =
  let key = key ~base_path ~keeper_name in
  match List.assoc_opt key (Atomic.get owners) with
  | None -> Owner_absent
  | Some owner -> Owner_projection (Keeper_owner.projection owner)
;;

end

module Candidate = struct
type lookup =
  | Owner_absent
  | Owner_projection of Keeper_owner_reducer.projection

module Owner_map = Set_util.StringMap

let owners : Keeper_owner.t Owner_map.t Atomic.t = Atomic.make Owner_map.empty

let cells_mu = Stdlib.Mutex.create ()

let key ~base_path ~keeper_name =
  Keeper_registry_types.canonical_base_path_exn base_path ^ "\000" ^ keeper_name
;;

let install ~base_path ~keeper_name owner =
  let key = key ~base_path ~keeper_name in
  Stdlib.Mutex.protect cells_mu (fun () ->
    Atomic.set owners (Owner_map.add key owner (Atomic.get owners)))
;;

let remove ~base_path ~keeper_name owner =
  let key = key ~base_path ~keeper_name in
  Stdlib.Mutex.protect cells_mu (fun () ->
    let installed = Atomic.get owners in
    match Owner_map.find_opt key installed with
    | Some current when current == owner ->
        Atomic.set owners (Owner_map.remove key installed)
    | Some _ | None -> ())
;;

let lookup ~base_path ~keeper_name =
  let key = key ~base_path ~keeper_name in
  match Owner_map.find_opt key (Atomic.get owners) with
  | None -> Owner_absent
  | Some owner -> Owner_projection (Keeper_owner.projection owner)
;;

end

type api =
  { role : string
  ; install : base_path:string -> keeper_name:string -> Keeper_owner.t -> unit
  ; remove : base_path:string -> keeper_name:string -> Keeper_owner.t -> unit
  ; read : base_path:string -> keeper_name:string -> int
  }
let baseline_read ~base_path ~keeper_name =
  match Baseline.lookup ~base_path ~keeper_name with
  | Owner_absent -> -2
  | Owner_projection None -> -1
  | Owner_projection (Some value) -> value
let candidate_read ~base_path ~keeper_name =
  match Candidate.lookup ~base_path ~keeper_name with
  | Owner_absent -> -2
  | Owner_projection None -> -1
  | Owner_projection (Some value) -> value
let baseline =
  { role = "baseline"; install = Baseline.install; remove = Baseline.remove; read = baseline_read }
let candidate =
  { role = "candidate"; install = Candidate.install; remove = Candidate.remove; read = candidate_read }
let roles = [baseline; candidate]
let owner value = Keeper_owner.{ state = Atomic.make value }
let checks = ref 0
let check expected actual =
  incr checks;
  if expected <> actual then
    failwith (Printf.sprintf "semantic check expected %d, actual %d" expected actual)
